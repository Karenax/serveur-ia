cmake_minimum_required(VERSION 3.22)
project(fred_voice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded" CACHE STRING "" FORCE)
endif()
if(NOT WHISPER_SOURCE_DIR)
  message(FATAL_ERROR "WHISPER_SOURCE_DIR must identify the pinned whisper.cpp source")
endif()
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
if(ANDROID)
  if(ANDROID_ABI STREQUAL "x86_64")
    # Probe optional SIMD at runtime, retaining a baseline for older CPUs.
    set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
  else()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
  endif()
  add_link_options("-Wl,-z,max-page-size=16384")
else()
  set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
  set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
endif()
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper)
add_library(fredvoice SHARED fred_voice.cpp)
if(WIN32)
  add_library(fredaec SHARED fred_aec.cpp)
  target_link_libraries(fredaec PRIVATE ole32 uuid msdmo dmoguids strmiids wmcodecdspuuid propsys)
endif()
target_link_libraries(fredvoice PRIVATE whisper)
target_include_directories(fredvoice PRIVATE "${WHISPER_SOURCE_DIR}/include" "${WHISPER_SOURCE_DIR}/ggml/include")
if(ANDROID)
  target_sources(fredvoice PRIVATE fred_voice_jni.cpp)
  target_link_options(fredvoice PRIVATE "-Wl,-z,max-page-size=16384")
endif()
